cmake_minimum_required(VERSION 3.20)
project(PrologWinDbg VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)

add_executable(protocol_tests tests/protocol_tests.cpp)
target_include_directories(protocol_tests PRIVATE src)
add_test(NAME protocol COMMAND protocol_tests)

find_program(SWIPL_EXECUTABLE NAMES swipl swipl.exe)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(SWIPL_EXECUTABLE)
  add_test(NAME prolog_rules COMMAND "${SWIPL_EXECUTABLE}" -q -f none
    -s "${CMAKE_CURRENT_SOURCE_DIR}/tests/rules_tests.pl" -g run_tests -t halt)
  if(Python3_Interpreter_FOUND)
    add_test(NAME worker_integration COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_worker.py" "${SWIPL_EXECUTABLE}")
  endif()
else()
  message(STATUS "SWI-Prolog not found: Prolog and worker tests are not registered")
endif()
if(NOT Python3_Interpreter_FOUND)
  message(STATUS "Python 3 not found: worker protocol integration tests are not registered")
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Version 1 requires an x64 build.")
  endif()
  set(DEBUGGING_TOOLS_ROOT "" CACHE PATH "Windows Kits/10/Debuggers directory")
  add_library(prolog SHARED src/extension.cpp src/debugger.cpp src/worker_process.cpp src/guids.cpp src/prolog.def)
  target_compile_definitions(prolog PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_include_directories(prolog PRIVATE src)
  if(DEBUGGING_TOOLS_ROOT)
    target_include_directories(prolog PRIVATE "${DEBUGGING_TOOLS_ROOT}/sdk/inc" "${DEBUGGING_TOOLS_ROOT}/inc")
  endif()
  set_target_properties(prolog PROPERTIES PREFIX "" OUTPUT_NAME "prolog")
  add_executable(prologdbg_target tests/windows/test_target.cpp)
  target_compile_definitions(prologdbg_target PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(prologdbg_target PRIVATE dbghelp)
  if(MSVC)
    target_compile_options(prolog PRIVATE /W4 /EHsc /utf-8 /permissive-)
    target_compile_options(prologdbg_target PRIVATE /W4 /EHsc /Zi /Od /utf-8)
    target_link_options(prologdbg_target PRIVATE /DEBUG /OPT:NOREF /OPT:NOICF)
  else()
    target_compile_options(prolog PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(prologdbg_target PRIVATE -g -O0)
    target_link_options(prolog PRIVATE -static-libgcc -static-libstdc++ -static)
    target_link_options(prologdbg_target PRIVATE -static-libgcc -static-libstdc++ -static)
  endif()
  install(TARGETS prolog prologdbg_target RUNTIME DESTINATION . LIBRARY DESTINATION .)
  install(DIRECTORY prolog examples DESTINATION .)
  install(FILES README.md THIRD_PARTY_NOTICES.txt DESTINATION .)
  install(DIRECTORY docs licenses DESTINATION .)
endif()
